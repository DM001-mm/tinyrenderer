网络流最大流算法模板

1. Ford-Fulkerson 算法（DFS 版本）
==========================================

// DFS 求增广路径
bool dfs(int u, int sink, int pushed) {
    if (u == sink) return true;
    visited[u] = true;

    for (int v : graph[u]) {
        if (!visited[v] && capacity[u][v] > 0) {
            int flow = min(pushed, capacity[u][v]);
            if (dfs(v, sink, flow)) {
                capacity[u][v] -= flow;
                capacity[v][u] += flow;
                return true;
            }
        }
    }
    return false;
}

int maxFlow(int source, int sink, int n) {
    int flow = 0;
    while (true) {
        memset(visited, false, sizeof(visited));
        if (!dfs(source, sink, INF)) break;
        flow += 找到的流量;
    }
    return flow;
}

2. Dinic 算法（分层图）
==========================================

vector<int> level, iter;

bool bfs(int source, int sink, int n) {
    level.assign(n, -1);
    level[source] = 0;
    queue<int> q;
    q.push(source);

    while (!q.empty()) {
        int u = q.front();
        q.pop();

        for (auto& edge : graph[u]) {
            int v = edge.to;
            if (level[v] < 0 && edge.cap > 0) {
                level[v] = level[u] + 1;
                q.push(v);
            }
        }
    }
    return level[sink] >= 0;
}

int dfs(int u, int sink, int flow) {
    if (u == sink) return flow;

    for (int& i = iter[u]; i < graph[u].size(); i++) {
        Edge& e = graph[u][i];
        if (level[u] < level[e.to] && e.cap > 0) {
            int d = dfs(e.to, sink, min(flow, e.cap));
            if (d > 0) {
                e.cap -= d;
                graph[e.to][e.rev].cap += d;
                return d;
            }
        }
    }
    return 0;
}

int maxFlow(int source, int sink, int n) {
    int flow = 0;
    while (bfs(source, sink, n)) {
        iter.assign(n, 0);
        int f;
        while ((f = dfs(source, sink, INF)) > 0) {
            flow += f;
        }
    }
    return flow;
}

3. Push-Relabel 算法
==========================================

struct Edge {
    int to, cap, rev;
};

vector<vector<Edge>> graph;
vector<int> h, e;

void push(int u, Edge& e) {
    int d = min(e.cap, e[u] - e[e.to]);
    e.cap -= d;
    graph[e.to][e.rev].cap += d;
    e[e.to] += d;
    e[u] -= d;
}

void relabel(int u) {
    int minH = INT_MAX;
    for (auto& e : graph[u]) {
        if (e.cap > 0) {
            minH = min(minH, h[e.to]);
        }
    }
    h[u] = minH + 1;
}

int maxFlow(int source, int sink, int n) {
    h.assign(n, 0);
    e.assign(n, 0);
    h[source] = n;

    queue<int> q;
    for (auto& edge : graph[source]) {
        edge.cap = edge.cap;  // 初始化流
        if (edge.cap > 0) {
            push(source, edge);
            if (edge.to != sink) q.push(edge.to);
        }
    }

    while (!q.empty()) {
        int u = q.front();
        q.pop();

        while (e[u] > 0) {
            bool pushed = false;
            for (auto& edge : graph[u]) {
                if (edge.cap > 0 && h[u] == h[edge.to] + 1) {
                    push(u, edge);
                    if (edge.to != sink) q.push(edge.to);
                    pushed = true;
                    break;
                }
            }
            if (!pushed) {
                relabel(u);
            }
        }
    }

    return e[sink];
}

4. 最小费用最大流（MCMF）
==========================================

struct Edge {
    int to, cap, cost, flow;
};

vector<vector<int>> g;
vector<Edge> edges;

void addEdge(int u, int v, int cap, int cost) {
    g[u].push_back(edges.size());
    edges.push_back({v, cap, cost, 0});
    g[v].push_back(edges.size());
    edges.push_back({u, 0, -cost, 0});
}

pair<int, int> minCostMaxFlow(int source, int sink, int n) {
    int flow = 0, cost = 0;
    vector<int> dist(n);
    vector<int> parent(n);
    vector<int> parentEdge(n);

    while (true) {
        dist.assign(n, INT_MAX);
        dist[source] = 0;

        // SPFA 算法
        queue<int> q;
        q.push(source);
        vector<bool> inQueue(n, false);
        inQueue[source] = true;

        while (!q.empty()) {
            int u = q.front();
            q.pop();
            inQueue[u] = false;

            for (int idx : g[u]) {
                Edge& e = edges[idx];
                if (e.cap > e.flow && dist[e.to] > dist[u] + e.cost) {
                    dist[e.to] = dist[u] + e.cost;
                    parent[e.to] = u;
                    parentEdge[e.to] = idx;
                    if (!inQueue[e.to]) {
                        q.push(e.to);
                        inQueue[e.to] = true;
                    }
                }
            }
        }

        if (dist[sink] == INT_MAX) break;

        int pushFlow = INT_MAX;
        for (int u = sink; u != source; u = parent[u]) {
            pushFlow = min(pushFlow, edges[parentEdge[u]].cap - edges[parentEdge[u]].flow);
        }

        flow += pushFlow;
        cost += pushFlow * dist[sink];

        for (int u = sink; u != source; u = parent[u]) {
            edges[parentEdge[u]].flow += pushFlow;
            edges[parentEdge[u] ^ 1].flow -= pushFlow;
        }
    }

    return {flow, cost};
}

==========================================
注意事项：
1. 容量必须为非负整数
2. 源点和汇点不能相同
3. 选择合适的算法：
   - 一般情况：Dinic（时间复杂度 O(V²E)）
   - 稀疏图：Ford-Fulkerson + BFS（O(VE²)）
   - 稠密图：Push-Relabel（O(V³)）
   - 需要最小费用：MCMF
4. 处理多源多汇时，添加虚拟源点和汇点
